package io.github.broqlinq.jaguar.reactive.internal;

import java.util.Arrays;

final class ConsumerList {

    private final Producer producer;

    private Link[] links = Link.EMPTY_LIST;
    private int size = 0;

    public ConsumerList(Producer producer) {
        this.producer = producer;
    }

    public void add(Link link) {
        if (updateIfPresent(link)) {
            return;
        }

        final int len = links.length;
        final int index = this.size;

        if (len == 0) {
            links = new Link[2];
        } else if (index == len) {
            int growth = len < 64 ? len : len >> 1;
            links = Arrays.copyOf(links, len + growth);
        }

        links[index] = link;
        this.size = index + 1;
        link.producer = producer;
        link.indexInProducer = index;
    }

    private boolean updateIfPresent(Link link) {
        final int index = link.indexInProducer;
        if (index == -1) {
            return false;
        }
        if (links[index] != link) {
            throw new IllegalArgumentException("Corrupted link list in producer");
        }
        link.markSeen();
        return true;
    }

    public void remove(Link link) {
        final int size = this.size;
        final int index = link.indexInProducer;
        var removed = links[index];
        if (removed != link) {
            throw new IllegalArgumentException("Corrupted link list in producer");
        }
        var moved = links[size - 1];

        links[size - 1] = null;

        if (index != size - 1) {
            links[index] = moved;
            moved.indexInProducer = index;
        }

        this.size = size - 1;
    }

    public boolean nonEmpty() {
        return size != 0;
    }

    public void notifyDependents() {
        for (int i = 0; i < size; i++) {
            links[i].invalidate();
        }
    }

    public void unlinkAll() {
        for (int i = size - 1; i >= 0; i--) {
            var link = links[i];
            link.unlink();
        }
    }
}

package io.github.broqlinq.jaguar.reactive.internal;

final class Link {

    static final Link[] EMPTY_LIST = new Link[0];

    Producer producer;
    Consumer consumer;

    long lastVersion;
    long lastEpoch;

    int indexInProducer;
    int indexInConsumer;

    boolean active;

    Link() {
        clear();
    }

    void invalidate() {
        if (producer.version() != lastVersion) {
            consumer.invalidate();
        }
    }

    void markSeen() {
        this.lastVersion = producer.version();
        this.lastEpoch = consumer.epoch();
    }

    public void link(Producer producer, Consumer consumer) {
        this.producer = producer;
        this.consumer = consumer;

        producer.consumerLinks().add(this);
        consumer.producerLinks().add(this);

        markSeen();

        active = true;
    }

    public void unlink() {
        if (!active) {
            return;
        }

        producer.consumerLinks().remove(this);
        consumer.producerLinks().remove(this);

        clear();
    }

    public void clear() {
        active = false;

        producer = null;
        consumer = null;

        lastVersion = -1;
        lastEpoch = -1;

        indexInProducer = -1;
        indexInConsumer = -1;
    }

    @Override
    public String toString() {
        return "Link[" + consumer + " (" + indexInConsumer + ") -> " + producer + " (" + indexInProducer + ")]";
    }
}

package io.github.broqlinq.jaguar.reactive.internal;

import java.util.ArrayDeque;
import java.util.Arrays;
import java.util.IdentityHashMap;

final class ProducerList {

    private final ArrayDeque<Link> pool = new ArrayDeque<>();
    private final IdentityHashMap<Producer, Link> producerLinks = new IdentityHashMap<>();
    private Link[] links = Link.EMPTY_LIST;
    private int size = 0;

    private final Consumer consumer;

    public ProducerList(Consumer consumer) {
        this.consumer = consumer;
        for (int i = 0; i < 128; i++) {
            pool.offer(new Link());
        }
    }

    public Link getOrCreate(Producer producer) {
        Link link;
        if (size > 32) {
            link = producerLinks.get(producer);
        } else {
            link = null;
            for (int i = 0; i < size; i++) {
                if (links[i].producer == producer) {
                    link = links[i];
                    break;
                }
            }
        }
        if (link != null) {
            return link;
        }

        link = pool.poll();
        if (link == null) {
            link = new Link();
        }
        return link;
    }

    public void add(Link link) {
        if (updateIfPresent(link)) {
            return;
        }

        final int len = links.length;
        final int size = this.size;

        if (size == len) {
            int length = len == 0 ? 4 : len + (len >> 1);
            links = Arrays.copyOf(links, length);
        }

        links[size] = link;
        this.size = size + 1;
        link.consumer = consumer;
        link.indexInConsumer = size;
        producerLinks.put(link.producer, link);
    }

    private boolean updateIfPresent(Link link) {
        final int index = link.indexInConsumer;
        if (index == -1) {
            return false;
        }
        if (links[index] != link) {
            throw new IllegalArgumentException("Corrupted link list in producer");
        }
        link.markSeen();
        return true;
    }

    public void remove(Link link) {
        final int size = this.size;
        final int index = link.indexInConsumer;
        var removed = links[index];
        if (removed != link) {
            throw new IllegalArgumentException("Corrupted link list in consumer: " + link);
        }
        var moved = links[size - 1];

        links[size - 1] = null;

        if (index != size - 1) {
            links[index] = moved;
            moved.indexInConsumer = index;
        }
        this.size = size - 1;
        producerLinks.remove(link.producer);
    }

    public void removeStale() {
        final int len = size;
        for (int i = len - 1; i >= 0; i--) {
            var link = links[i];
            if (link.lastEpoch != consumer.epoch()) {
//                links[i] = null;
                producerLinks.remove(link.producer);
                link.unlink();
                pool.offer(link);
            }
        }

    }

    public void unlinkAll() {
        for (int i = size - 1; i >= 0; i--) {
            links[i].unlink();
        }
    }
}

        disposed = true;
        onDispose();

        scope.assertOwnerThread();

        if (isDisposed()) {
            throw new IllegalStateException("Attempted read from disposed node");
        }
    }

    protected final void assertWritable() {
        scope.assertOwnerThread();

        if (isDisposed()) {
            throw new IllegalStateException("Attempted write to disposed node");
        }
    }

package io.github.broqlinq.jaguar.reactive.internal;

import java.util.Objects;

import java.util.function.Supplier;

final class ScopedComputed<T> extends ScopeOwned implements Signal<T>, Producer, Consumer, SchedulerTask {

    private final ProducerList producerLinks = new ProducerList(this);
    private final ConsumerList consumerLinks = new ConsumerList(this);

    private final Supplier<? extends T> supplier;
    private T cachedValue = null;

    private boolean invalidated = true;
    private boolean disposed = false;
    private long version = 0;
    private long epoch = 0;
    private boolean queued = false;
    private boolean tracking;

    ScopedComputed(ThreadConfinedScope context, Supplier<? extends T> supplier) {
        this.supplier = Objects.requireNonNull(supplier, "supplier");
        super(context);
    }

    @Override
    public T value() {
        assertReadable("Attempted read from disposed signal");

        scope.trackingContext().track(this);

        recomputeIfNeeded();

        return cachedValue;
    }

    private void recomputeIfNeeded() {
        if (!invalidated) {
            return;
        }

        epoch++;
        T newValue;
        scope.trackingContext().push(this);
        try {
            newValue = supplier.get();
        } finally {
            scope.trackingContext().pop(this);
        }

        // TODO: Do proper equality check, identity check for now
        if (cachedValue != newValue) {
            cachedValue = newValue;
            version++;
        }

        producerLinks.removeStale();

        invalidated = false;

        scope.batch(this);
    }

    @Override
    public <R> Signal<R> derive(Function<? super T, ? extends R> mapper) {
        Objects.requireNonNull(mapper, "mapper");
        assertReadable("Attempted read from disposed signal");
        return scope.computed(() -> mapper.apply(value()));
    }

    @Override
    public <R> R transform(Function<? super T, ? extends R> mapper) {
        Objects.requireNonNull(mapper, "mapper");
        assertReadable("Attempted read from disposed signal");
        return mapper.apply(value());
    }

    @Override
    public void dispose() {
        if (disposed) {
            return;
        }
        disposed = true;

        consumerLinks.unlinkAll();
        scope.removeComputed(this);
    }

    @Override
    public boolean isDisposed() {
        return disposed || scope.isClosed();
    }

    @Override
    public long version() {
        return version;
    }

    @Override
    public ConsumerList consumerLinks() {
        return consumerLinks;
    }

    @Override
    public void invalidate() {
        if (invalidated) {
            return;
        }
        invalidated = true;
        queued = true;
        scope.batch(this);
    }

    @Override
    public long epoch() {
        return epoch;
    }

    @Override
    public ProducerList producerLinks() {
        return producerLinks;
    }

    @Override
    public boolean isTracking() {
        return tracking;
    }

    @Override
    public void setTracking(boolean tracking) {
        this.tracking = tracking;
    }

    @Override
    public void execute() {
        queued = false;
        consumerLinks.notifyDependents();
    }
}

package io.github.broqlinq.jaguar.reactive.internal;

import io.github.broqlinq.jaguar.reactive.Effect;

import java.util.Objects;

final class ScopedEffect extends ConsumerNode implements Effect {

    private final QueueableEffect effect;

    ScopedEffect(ThreadConfinedScope context, Runnable effect) {
        super(context);
        Objects.requireNonNull(effect, "effect");
        this.effect = new QueueableEffect(effect);
        this.invalidated = true;
    }

    @Override
    public void invalidate() {
        if (invalidated) {
            return;
        }
        invalidated = true;
        scheduleEffect();
    }

    @Override
    protected void onDispose() {
        invalidated = false;
        producerLinks.unlinkAll();
        scope.removeEffect(this);
    }

    void scheduleEffect() {
        if (isDisposed() || effect.queued) {
            return;
        }

        effect.queued = true;
        scope.schedule(effect);
    }

    private final class QueueableEffect implements Runnable {
        private final Runnable runnable;
        private boolean queued = false;

        QueueableEffect(Runnable effect) {
            this.runnable = effect;
        }

        @Override
        public void run() {
            queued = false;

            if (!invalidated || isDisposed()) {
                return;
            }

            updateEpoch();
            boolean completed = false;
            scope.trackingContext().push(ScopedEffect.this);
            try {
                runnable.run();
                completed = true;
            } finally {
                scope.trackingContext().pop(ScopedEffect.this);
                if (completed) {
                    producerLinks.removeStale();
                    invalidated = false;
                }
            }
        }
    }
}

final class ScopedIntComputed extends ProducerConsumerNode implements IntSignal {

    private final Runnable batchTask = consumerLinks::notifyDependents;

    private final IntSupplier supplier;
    private int cachedValue;

    ScopedIntComputed(ThreadConfinedScope scope, IntSupplier supplier) {
        this.supplier = Objects.requireNonNull(supplier, "supplier");
        super(scope);
        this.invalidated = true;
    }

    @Override
    public int intValue() {
        assertReadable();
        return valueUnchecked();
    }

    private int valueUnchecked() {
        scope.trackingContext().track(this);
        if (invalidated) {
            recompute();
        }
        return cachedValue;
    }

    private void recompute() {
        updateEpoch();
        int newValue;
        scope.trackingContext().push(this);
        try {
            newValue = supplier.getAsInt();
        } finally {
            scope.trackingContext().pop(this);
        }
        final int oldValue = cachedValue;
        if (oldValue != newValue) {
            cachedValue = newValue;
            updateVersion();
        }

        producerLinks.removeStale();
        invalidated = false;

        batchNotifyDependents();
    }

    private void batchNotifyDependents() {
        if (consumerLinks().nonEmpty()) {
            scope.batch(batchTask);
        }
    }

    @Override
    public <R> Signal<R> derive(IntFunction<? extends R> mapper) {
        assertReadable();
        Objects.requireNonNull(mapper, "mapper");
        return scope.computed(() -> mapper.apply(valueUnchecked()));
    }

    public IntSignal deriveInt(IntUnaryOperator mapper) {
        assertReadable();
        Objects.requireNonNull(mapper, "mapper");
        return scope.computedInt(() -> mapper.applyAsInt(valueUnchecked()));
    }

    @Override
    public void invalidate() {
        if (invalidated) {
            return;
        }
        invalidated = true;
        batchNotifyDependents();
    }

    @Override
    protected void onDispose() {
        invalidated = false;
        consumerLinks.unlinkAll();
        producerLinks.unlinkAll();
        scope.removeComputed(this);
    }

import java.util.function.Function;

final class ScopedState<T> extends ProducerNode implements State<T> {

    private final Runnable notifyTask = consumerLinks::notifyDependents;

    private final BiPredicate<? super T, ? super T> eq;
    private T value;

    ScopedState(ThreadConfinedScope scope, T initialValue, BiPredicate<? super T, ? super T> eq) {
        this.value = initialValue;
        this.eq = Objects.requireNonNull(eq, "eq");
        super(scope);

        if (eq.test(oldValue, newValue)) {
            return;
        }
        this.value = newValue;
        updateVersion();
        batchNotifyDependents();
    }

    private void batchNotifyDependents() {
        if (consumerLinks().nonEmpty()) {
            scope.batch(notifyTask);
        }
    }

    @Override
    public T value() {
        assertReadable();
        return valueUnchecked();
    }

    T valueUnchecked() {
        scope.trackingContext().track(this);
        return value;

        assertReadable();
        Objects.requireNonNull(mapper, "mapper");
        return scope.computed(() -> mapper.apply(valueUnchecked()));

abstract class ScopeOwned implements Scoped {

    protected final ThreadConfinedScope scope;
    protected final List<Runnable> cleanup = new ArrayList<>(1);

    public ScopeOwned(ThreadConfinedScope scope) {
        this.scope = Objects.requireNonNull(scope, "context");
    }

    @Override
    public Scope scope() {
        return scope;
    }

    public boolean isDisposed() {
        return scope.isClosed();
    }

    protected void assertNotDisposed(String message) {
        if (isDisposed()) {
            throw new IllegalStateException(message);
        }
    }

    protected void assertReadable(String message) {
        scope.assertOwnerThread();
        assertNotDisposed(message);
    }

    protected void assertWritable(String message) {
        scope.assertOwnerThread();
        assertNotDisposed(message);
    }

    protected final void onDispose(Runnable cleanup) {
        this.cleanup.add(cleanup);
    }

    protected final void cleanup() {
        Throwable fail = null;
        for (var cleanup : cleanup) {
            try {
                cleanup.run();
            } catch (Throwable ex) {
                if (fail == null) {
                    fail = ex;
                } else {
                    fail.addSuppressed(ex);
                }
            }
        }
        if (fail != null) {
            throw new RuntimeException(fail);
        }
    }
}

package io.github.broqlinq.jaguar.reactive.internal;

import io.github.broqlinq.jaguar.reactive.Scheduler;
import io.github.broqlinq.jaguar.reactive.SchedulerTask;

import java.util.ArrayDeque;

public final class ThreadConfinedScheduler implements Scheduler {

    private final Thread ownerThread;
    private final ArrayDeque<Runnable> tasks = new ArrayDeque<>(32);
    private int batchDepth = 0;
    private boolean flushing = false;

    public ThreadConfinedScheduler(Thread ownerThread) {
        this.ownerThread = ownerThread;
    }

    @Override
    public boolean isOwnerThread() {
        return Thread.currentThread() == ownerThread;
    }

    @Override
    public void schedule(Runnable task) {
        assertOwnerThread("Attempted to schedule task on non-owner thread");

        tasks.offerLast(task);

        if (batchDepth == 0) {
            flush();
        }
    }

    @Override
    public void batch(Runnable task) {
        assertOwnerThread("Attempted to batch task on non-owner thread");

        batchDepth++;

        boolean completed = false;
        try {
            task.run();
            completed = true;
        } finally {
            batchDepth--;
            if (batchDepth == 0 && completed) {
                flush();
            }
        }
    }

    private void assertOwnerThread(String message) {
        if (!isOwnerThread()) {
            throw new WrongThreadException(message);
        }
    }

    private void flush() {
        if (flushing) {
            return;
        }

        flushing = true;
        try {
            Runnable task;
            while ((task = tasks.pollFirst()) != null) {
                task.run();
            }
        } finally {
            flushing = false;
        }
    }
}

package io.github.broqlinq.jaguar.reactive.internal;

import java.util.Arrays;

final class TrackingContext {

    private Consumer[] stack = new Consumer[128];
    private int size = 0;

    private Consumer peek() {
        if (size == 0) {
            return null;
        }
        return stack[size - 1];
    }

    public void track(Producer producer) {
        if (size > 0) {
            var consumer = stack[size - 1];
            var link = consumer.producerLinks().getOrCreate(producer);
            if (link.active) {
                link.markSeen();
            } else {
                link.link(producer, consumer);
            }
        }
    }

    public void push(Consumer consumer) {
        if (consumer.isTracking()) {
            throw new IllegalStateException("Consumer is already in tracking context, circular dependency detected");
        }
        if (size == stack.length) {
            int newSize = stack.length + stack.length / 2;
            stack = Arrays.copyOf(stack, newSize);
        }
        stack[size++] = consumer;
        consumer.setTracking(true);
    }

    public void pop(Consumer expected) {
        if (size == 0 || stack[size - 1] != expected) {
            throw new IllegalStateException("Tracking stack corrupted");
        }

        Consumer consumer = stack[--size];
        stack[size] = null;
        consumer.setTracking(false);
    }
}
